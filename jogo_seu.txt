                player.velocityY = jumpPower;

            }

                   obj1.y + obj1.height > obj2.y;
        }

        function update() {
            if (!gameRunning || isPlayingAnimation) return;
            
            frameCount++;
            
            if (score >= 1625 && !boss && !icePlanet) {
                boss = {
                    x: 650,
                    y: 260,
                    width: 60,
                    height: 80,
                    hits: 3,
                    defeated: false
                };
                player.canMove = true;
                gameSpeed = 0;
                obstacles = [];
                coins = [];
            }
            
            if (boss && player.canMove) {
                player.velocityX = 0;
                if (keys['ArrowLeft'] || keys['a']) player.velocityX = -moveSpeed;
                if (keys['ArrowRight'] || keys['d']) player.velocityX = moveSpeed;
                player.x += player.velocityX;
                if (player.x < 0) player.x = 0;
                if (player.x > canvas.width - player.width) player.x = canvas.width - player.width;
            }
            
            if (boss && !boss.defeated && frameCount % 120 === 0) {
                rocks.push({
                    x: boss.x + 25,
                    y: boss.y + 40,
                    width: 20,
                    height: 20,
                    velocityY: 0,
                    velocityX: 0,
                    gravity: 0.5,
                    isLaser: false
                });
            }
            
            for (let i = rocks.length - 1; i >= 0; i--) {
                const rock = rocks[i];
                
                if (rock.isLaser) {
                    // Laser se move horizontalmente
                    rock.x += rock.velocityX;
                } else {
                    // Pedra cai (do chimpanzé)
                    rock.velocityY += rock.gravity;
                    rock.y += rock.velocityY;
                    
                    if (rock.y >= ground + player.height - rock.height) {
                        rock.y = ground + player.height - rock.height;
                        rock.velocityY = 0;
                    }
                }
                
                if (boss && !boss.defeated && rock.isLaser && checkCollision(rock, boss)) {
                    // Só conta se for LASER do jogador, não pedra do boss
                    boss.hits--;
                    rocks.splice(i, 1);
                    
                    // Efeito visual de acerto
                    boss.x += (Math.random() - 0.5) * 8;
                    
                    console.log("Boss atingido por LASER! Vidas restantes: " + boss.hits);
                    
                    if (boss.hits <= 0) {
                        console.log("Boss derrotado! Iniciando animação...");
                        boss.defeated = true;
                        score += 500;
                        updateScore();
                        playBossDefeatAnimation();
                    }
                    continue;
                }
                
                // Colisão da pedra do boss com o jogador
                if (!rock.isLaser && checkCollision(rock, player)) {
                    lives--;
                    updateLives();
                    rocks.splice(i, 1);
                    if (lives <= 0) gameOver();
                    continue;
                }
                
                // Remover se sair da tela
                if (rock.y > canvas.height || rock.x > canvas.width || rock.x < -50) {
                    rocks.splice(i, 1);
                }
            }
            
            if (score >= 876 && score < 1134 && !ufo && !boss) {
                ufo = { x: canvas.width / 2, y: 80, velocityX: 2 };
            }
            
            if (score >= 1134 && ufo) {
                ufo = null;
                lasers = [];
            }
            
            if (ufo) {
                ufo.x += ufo.velocityX;
                if (ufo.x > canvas.width - 100 || ufo.x < 100) {
                    ufo.velocityX *= -1;
                }
                if (frameCount % 120 === 0) {
                    lasers.push({ x: ufo.x, y: ufo.y + 30, width: 4, height: 20, speed: 6 });
                }
            }
            
            for (let i = lasers.length - 1; i >= 0; i--) {
                const laser = lasers[i];
                laser.y += laser.speed;
                
                if (checkCollision(player, laser)) {
                    lives--;
                    updateLives();
                    lasers.splice(i, 1);
                    if (lives <= 0) gameOver();
                }
                
                if (laser.y > canvas.height) lasers.splice(i, 1);
            }
            
            player.velocityY += gravity;
            player.y += player.velocityY;
            
            if (player.y >= ground) {
                player.y = ground;
                player.velocityY = 0;
                player.jumping = false;
            }
            
            if (frameCount % 120 === 0 && !boss) createObstacle();
            if (frameCount % 80 === 0 && !boss && !icePlanet) createCoin();
            
            if (!boss) {
                for (let i = obstacles.length - 1; i >= 0; i--) {
                    const obs = obstacles[i];
                    obs.x -= gameSpeed;
                    
                    const stompedOnHead = player.velocityY > 0 && 
                                          player.x < obs.x + obs.width &&
                                          player.x + player.width > obs.x &&
                                          player.y + player.height >= obs.y &&
                                          player.y + player.height <= obs.y + 10 &&
                                          (obs.type === 'goomba' || obs.type === 'skeleton' || obs.type === 'zombie');
                    
                    if (stompedOnHead) {
                        obstacles.splice(i, 1);
                        score += 30;
                        updateScore();
                        player.velocityY = jumpPower * 0.6;
                        continue;
                    }
                    
                    if (checkCollision(player, obs)) {
                        lives--;
                        updateLives();
                        obstacles.splice(i, 1);
                        if (lives <= 0) gameOver();
                    }
                    
                    if (obs.x + obs.width < 0) {
                        obstacles.splice(i, 1);
                        score += 10;
                        updateScore();
                    }
                }
            }
            
            for (let i = coins.length - 1; i >= 0; i--) {
                const coin = coins[i];
                coin.x -= gameSpeed;
                
                if (!coin.collected && checkCollision(player, coin)) {
                    coin.collected = true;
                    score += 50;
                    updateScore();
                }
                
                if (coin.x + coin.width < 0) coins.splice(i, 1);
            }
            
            if (frameCount % 500 === 0 && gameSpeed < 8) gameSpeed += 0.5;
        }
        
        function draw() {
            if (isPlayingAnimation) return;
            
            drawBackground();
            
            if (!boss) {
                obstacles.forEach(obs => drawObstacle(obs));
                coins.forEach(coin => drawCoin(coin));
            }
            
            rocks.forEach(rock => {
                if (rock.isLaser) {
                    // Desenhar laser azul
                    ctx.fillStyle = '#00ffff';
                    ctx.shadowBlur = 15;
                    ctx.shadowColor = '#00ffff';
                    ctx.fillRect(rock.x, rock.y, rock.width, rock.height);
                    ctx.shadowBlur = 0;
                    
                    // Rastro do laser
                    ctx.fillStyle = 'rgba(0, 255, 255, 0.3)';
                    ctx.fillRect(rock.x - 10, rock.y, 10, rock.height);
                } else {
                    // Pedra (do chimpanzé)
                    ctx.fillStyle = '#808080';
                    ctx.beginPath();
                    ctx.arc(rock.x + 10, rock.y + 10, 10, 0, Math.PI * 2);
                    ctx.fill();
                    ctx.fillStyle = '#606060';
                    ctx.beginPath();
                    ctx.arc(rock.x + 12, rock.y + 8, 4, 0, Math.PI * 2);
                    ctx.fill();
                }
            });
            
            lasers.forEach(laser => {
                ctx.fillStyle = '#00ff00';
                ctx.shadowBlur = 10;
                ctx.shadowColor = '#00ff00';
                ctx.fillRect(laser.x - 2, laser.y, laser.width, laser.height);
                ctx.shadowBlur = 0;
            });
            
            drawPlayer();
            
            if (boss && boss.hits > 0 && !boss.defeated) {
                ctx.fillStyle = 'rgba(0, 0, 0, 0.8)';
                ctx.fillRect(180, 130, 440, 150);
                ctx.fillStyle = '#fff';
                ctx.font = 'bold 26px Arial';
                ctx.textAlign = 'center';
                ctx.fillText('🦍 BOSS: CHIMPANZÉ! 🦍', 400, 170);
                ctx.font = '18px Arial';
                ctx.fillText('SETAS ← → para MOVER', 400, 205);
                ctx.fillStyle = '#00ffff';
                ctx.font = 'bold 18px Arial';
                ctx.fillText('ESPAÇO = PULAR + ATIRAR LASER! 🔫', 400, 235);
                ctx.fillStyle = '#ffff00';
                ctx.font = '16px Arial';
                ctx.fillText('⚡ Vida do Boss: ' + boss.hits + '/3 ⚡', 400, 265);
            }
            
            if (icePlanet && frameCount < 200) {
                ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
                ctx.fillRect(150, 120, 500, 100);
                ctx.fillStyle = '#00ffff';
                ctx.font = 'bold 26px Arial';
                ctx.textAlign = 'center';
                ctx.fillText('❄️ PLANETA DE GELO! ❄️', 400, 155);
                ctx.font = '16px Arial';
                ctx.fillText('Novo mundo! Novos obstáculos!', 400, 185);
                ctx.fillText('Blordon agora é uma nave! 🚀', 400, 210);
            }
        }
        
        function gameLoop() {
            if (!isPlayingAnimation) {
                update();
                draw();
            }
            if (gameRunning || isPlayingAnimation) {
                requestAnimationFrame(gameLoop);
            }
        }
        
        function updateScore() {
            document.getElementById('score').textContent = score;
            if (Math.floor(score / 500) > lastLifeBonus) {
                lastLifeBonus = Math.floor(score / 500);
                lives++;
                updateLives();
                showLifeBonus();
            }
        }
        
        function showLifeBonus() {
            const msg = document.createElement('div');
            msg.style.position = 'fixed';
            msg.style.top = '50%';
            msg.style.left = '50%';
            msg.style.transform = 'translate(-50%, -50%)';
            msg.style.background = 'rgba(255, 0, 0, 0.9)';
            msg.style.color = '#fff';
            msg.style.padding = '20px 40px';
            msg.style.borderRadius = '15px';
            msg.style.fontSize = '24px';
            msg.style.fontWeight = 'bold';
            msg.style.zIndex = '1000';
            msg.textContent = '❤️ +1 VIDA! ❤️';
            document.body.appendChild(msg);
            setTimeout(() => msg.remove(), 2000);
        }
        
        function updateLives() {
            let hearts = '';
            for (let i = 0; i < lives; i++) hearts += '❤️';
            document.getElementById('lives').textContent = hearts || '💀';
        }
        
        function gameOver() {
            gameRunning = false;
            document.getElementById('finalScore').textContent = score;
            document.getElementById('gameOver').style.display = 'block';
        }
        
        function restartGame() {
            gameRunning = true;
            score = 0;
            lives = 3;
            gameSpeed = 3;
            frameCount = 0;
            obstacles = [];
            coins = [];
            ufo = null;
            lasers = [];
            boss = null;
            rocks = [];
            keys = {};
            lastLifeBonus = 0;
            icePlanet = false;
            playerShip = false;
            isPlayingAnimation = false;
            player.x = 100;
            player.y = ground;
            player.velocityY = 0;
            player.velocityX = 0;
            player.jumping = false;
            player.canMove = false;
            
            updateScore();
            updateLives();
            document.getElementById('gameOver').style.display = 'none';
            gameLoop();
        }
        
        document.addEventListener('keydown', (e) => {
            keys[e.key] = true;
            if (e.code === 'Space' || e.code === 'ArrowUp') {
                e.preventDefault();
                jump();
            }
        });
        
        document.addEventListener('keyup', (e) => {
            keys[e.key] = false;
        });
        
        gameLoop();
    </script>
